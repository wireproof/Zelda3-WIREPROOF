cmake_minimum_required(VERSION 3.18)
project(Zelda3Wireproof VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
find_package(SDL2 CONFIG REQUIRED)
file(GLOB GAME_SOURCES CONFIGURE_DEPENDS src/*.c snes/*.c)
list(REMOVE_ITEM GAME_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/src/main.c")
add_library(zelda_engine OBJECT ${GAME_SOURCES}
  third_party/gl_core/gl_core_3_1.c
  third_party/opus-1.3.1-stripped/opus_decoder_amalgam.c)
target_include_directories(zelda_engine PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(zelda_engine PUBLIC SYSTEM_VOLUME_MIXER_AVAILABLE=0)
target_link_libraries(zelda_engine PUBLIC SDL2::SDL2)
if(NOT MSVC)
  target_link_libraries(zelda_engine PUBLIC m)
endif()
add_executable(zelda3 src/main.c)
target_link_libraries(zelda3 PRIVATE zelda_engine)
option(ZELDA_BUILD_SMOKE "Build desktop engine smoke harness" ON)
if(ZELDA_BUILD_SMOKE)
  add_executable(zelda3_smoke tests/engine_smoke.c)
  target_link_libraries(zelda3_smoke PRIVATE zelda_engine)
endif()
